☁️ Integrasi Database Online Google Drive (Google Sheets)

Hubungkan aplikasi ini dengan Google Apps Script di Google Drive Anda agar seluruh data Surat Tugas, SPD, SPJ, dan Master Pegawai tersimpan online dan bisa diakses dari HP atau komputer mana saja.

Tempelkan URL hasil Deploy Web App dari Google Sheets Database_ESPPD_Bapenda_Lahat Anda.

Panduan Setup Google Sheets & Kode Google Apps Script

  1. Buat Google Spreadsheet baru di Google Drive dengan nama Database_ESPPD_Bapenda_Lahat.
  2. Buat 4 tab sheet di bagian bawah dengan nama persis: Data_ST, Master_Pegawai, Master_Pejabat, dan Master_Tarif.
  3. Buka menu Ekstensi (Extensions) > Apps Script, hapus kode lama, lalu tempel kode di bawah ini.
  4. Klik Terapkan (Deploy) > Deployment baru, pilih jenis Aplikasi web, atur Jalankan sebagai: Saya (Me) dan Yang memiliki akses: Siapa saja (Anyone).
  5. Salin URL Web App yang muncul dan tempelkan pada kolom URL di atas.
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'getAll';
  if (action === 'getAll') {
    const result = {
      status: 'success',
      stList: readSheetJson(getOrCreateSheet(ss, 'Data_ST')),
      pegawaiList: readSheetJson(getOrCreateSheet(ss, 'Master_Pegawai')),
      pejabatList: readSheetJson(getOrCreateSheet(ss, 'Master_Pejabat')),
      tarifList: readSheetJson(getOrCreateSheet(ss, 'Master_Tarif'))
    };
    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const payload = JSON.parse(e.postData.contents);
  if (payload.action === 'syncAll') {
    writeSheetJson(getOrCreateSheet(ss, 'Data_ST'), payload.stList, 'id');
    writeSheetJson(getOrCreateSheet(ss, 'Master_Pegawai'), payload.pegawaiList, 'nip');
    writeSheetJson(getOrCreateSheet(ss, 'Master_Pejabat'), payload.pejabatList, 'id');
    writeSheetJson(getOrCreateSheet(ss, 'Master_Tarif'), payload.tarifList, 'id');
    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      message: 'Database Google Drive berhasil diperbarui',
      timestamp: new Date().toISOString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet(ss, name) {
  return ss.getSheetByName(name) || ss.insertSheet(name);
}

function readSheetJson(sheet) {
  if (!sheet) return [];
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  const values = sheet.getRange(2, 2, lastRow - 1, 1).getValues();
  const items = [];
  for (let i = 0; i < values.length; i++) {
    if (values[i][0]) {
      try { items.push(JSON.parse(values[i][0])); } catch (err) {}
    }
  }
  return items;
}

function writeSheetJson(sheet, dataArray, keyField) {
  if (!sheet) return;
  if (sheet.getLastRow() === 0) {
    sheet.getRange(1, 1, 1, 3).setValues([['ID_KEY', 'JSON_DATA', 'UPDATE_TERAKHIR']]);
  }
  const lastRow = sheet.getLastRow();
  if (lastRow >= 2) {
    sheet.getRange(2, 1, lastRow - 1, 3).clearContent();
  }
  if (!dataArray || dataArray.length === 0) return;
  const now = new Date().toLocaleString('id-ID');
  const rows = dataArray.map(function(item) {
    return [item[keyField] || '-', JSON.stringify(item), now];
  });
  sheet.getRange(2, 1, rows.length, 3).setValues(rows);
}
